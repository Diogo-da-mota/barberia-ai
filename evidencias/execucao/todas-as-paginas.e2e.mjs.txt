const browser = await chromium.launch({ channel: "chrome", headless: true })

async function open({ papel = null, width = 1440, height = 900 } = {}) {
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 768 })
  const page = await context.newPage()
  const errors = []
  const calls = []
  const state = { payments: [], savedProfile: null }
  page.on("pageerror", (e) => errors.push(e.message))
  page.on("console", (m) => { if (m.type() === "error" && !/Failed to load resource/.test(m.text())) errors.push(m.text()) })
  if (papel) {
    await context.addInitScript(({ s }) => {
      localStorage.setItem("barberia:theme", "dark")
      if (!sessionStorage.getItem("__s")) { sessionStorage.setItem("__s", "1"); localStorage.setItem("sb-hbkbdqctelembzqiptpi-auth-token", JSON.stringify(s)) }
    }, { s: SESSION })
  }
  await context.route(/supabase\.co/, async (route) => {
    const req = route.request()
    const url = new URL(req.url())
    const p = url.pathname
    const json = (b, st = 200) => route.fulfill({ status: st, contentType: "application/json", body: JSON.stringify(b) })
    const body = () => { try { return JSON.parse(req.postData() ?? "{}") } catch { return {} } }
    if (p.startsWith("/storage/v1/object/public/")) return route.fulfill({ status: 200, contentType: "image/png", body: PNG })
    if (p === "/auth/v1/user") return json(USER)
    if (p.includes("/rpc/")) {
      const fn = p.split("/rpc/")[1]
      calls.push({ fn, body: body() })
      switch (fn) {
        case "get_public_shop": return json(PUBLIC_SHOP)
        case "get_availability": return json([])
        case "get_team_links": return json([])
        case "my_professional": return json(papel === "barbeiro" ? "p1" : null)
        case "finance_report": return json([{ previsto_centavos: 18000, recebido_centavos: state.payments.reduce((t, x) => t + x.valor, 0), custos_centavos: 120000, comissoes_provisionadas_centavos: 2400, comissoes_liquidadas_centavos: 0, resultado_estimado_centavos: -122400 }])
        case "register_payment": { const b = body(); state.payments.push({ valor: b.p_valor_centavos, forma: b.p_forma }); return json("pay-1") }
        case "outbox_summary": return json([{ status: "bloqueado", total: 2 }, { status: "cancelado", total: 1 }, { status: "enviado", total: 0 }])
        case "get_plan_usage": return json([{ plano: "inicial", status: "trial", max_profissionais: 2, profissionais_ativos: 1, trial_ate: iso(now + 10 * 864e5), periodo_fim: null, permite_novos: true }])
        case "save_site_profile": state.savedProfile = body(); return json(null)
        default: return json(null)
      }
    }
    const t = p.replace("/rest/v1/", "")
    if (t.startsWith("memberships")) return json(papel === "barbeiro" && req.url().includes("papel=eq.barbeiro") ? [] : [{ papel, barbershops: SHOP }])
    if (t.startsWith("bookings")) return json(papel === "cliente" ? [FUTURE] : [BOOKING, FUTURE])
    if (t.startsWith("occupancies") || t.startsWith("invites") || t.startsWith("refunds") || t.startsWith("payouts") || t.startsWith("inbound_webhook_events") || t.startsWith("calendar_connections") || t.startsWith("customer_consents") || t.startsWith("working_hours")) return json([])
    if (t.startsWith("professionals")) return json([{ id: "p1", nome: "Ana", ativo: true, professional_services: [{ service_id: "s1" }] }])
    if (t.startsWith("services")) return json([{ id: "s1", nome: "Corte", descricao: null, preco_centavos: 6000, duracao_min: 40, ativo: true }])
    if (t.startsWith("payments")) return json(state.payments.map((x, i) => ({ id: `pay-${i}`, booking_id: "b1", valor_centavos: x.valor, forma: x.forma, parcelas: 1, criado_em: iso(now) })))
    if (t.startsWith("commission_rules")) return json([{ id: "r1", professional_id: null, service_id: null, tipo: "percentual", valor: 4000 }])
    if (t.startsWith("commissions")) return json([{ id: "c1", booking_id: "b1", professional_id: "p1", valor_centavos: 2400, base_centavos: 6000, regra_tipo: "percentual", regra_valor: 4000, status: "provisionada", criado_em: iso(now), professionals: { nome: "Ana" } }])
    if (t.startsWith("expenses")) return json([{ id: "e1", descricao: "Aluguel", categoria: "aluguel", valor_centavos: 120000, data: new Date(now).toISOString().slice(0, 10) }])
    if (t.startsWith("outbox_events")) return json([
      { id: "o1", tipo: "reserva_criada", canal: "whatsapp", destino: "11912345678", status: "bloqueado", tentativas: 0, max_tentativas: 5, ultimo_erro: "canal_nao_configurado", criado_em: iso(now - 600e3) },
      { id: "o2", tipo: "lembrete_24h", canal: "whatsapp", destino: "11987654321", status: "cancelado", tentativas: 0, max_tentativas: 5, ultimo_erro: "sem_consentimento", criado_em: iso(now - 900e3) },
    ])
    if (t.startsWith("integration_accounts")) return json([])
    if (t.startsWith("barbershops")) return json({ descricao: "Cortes clássicos desde 2010.", cor_destaque: "azul", endereco: "Rua de Teste, 10", telefone: "11999990000", slug: "barbearia-ficticia" })
    if (t.startsWith("media_objects")) return json([{ id: "m1", kind: "logo", storage_path: "loja-1/logo-a.png", mime: "image/png", tamanho_bytes: 1000, criado_em: iso(now) }])
    return route.abort()
  })
  return { page, context, errors, calls, state }
}

        main.scrollTop = 9999
        window.scrollTo(0, 9999)
        await new Promise((r) => setTimeout(r, 150))
        return { hasHeader: Boolean(header), hasAside: Boolean(aside), docScroll: document.scrollingElement.scrollHeight - innerHeight, winY: scrollY, antes, depois: { a: rect(aside), h: rect(header), b: rect(bottom) } }
      })
      const fixo = m.hasHeader && m.hasAside && m.docScroll <= 0 && m.winY === 0 && m.antes.a === m.depois.a && m.antes.h === m.depois.h && m.antes.b === m.depois.b
      check(`layout fixo ${w}x${h} ${path} [${papel}] : sidebar/header/barra inferior não se movem e o documento não rola`, fixo, JSON.stringify(m))
      await context.close()
    }
  }
}

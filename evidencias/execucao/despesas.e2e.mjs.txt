async function open({ width = 1440, height = 900 } = {}) {
  const browser = await chromium.launch({ channel: "chrome", headless: true })
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 768 })
  const page = await context.newPage()
  const errors = [], calls = []
  const state = { entries: [conta("d1", "Internet", "Internet", 9990, d(-3), { forma: "pix" }), conta("d2", "Energia elétrica", "Energia", 38000, d(0), { forma: "boleto", linha_digitavel: "34191.79001 01043.510047 91020.150008 5 90000000038000" }), conta("d3", "Aluguel", "Aluguel", 150000, d(-6), { situacao: "paga", pago_em: d(-6), valor_pago_centavos: 150000, forma: "pix", recurrence_id: "r1" })] }
  page.on("pageerror", (e) => errors.push(e.message))
  page.on("console", (m) => { if (m.type() === "error" && !/Failed to load resource/.test(m.text())) errors.push(m.text()) })
  await context.addInitScript(({ s }) => { localStorage.setItem("barberia:theme", "dark"); localStorage.setItem("barberia-auth-painel", JSON.stringify(s)) }, { s: SESSION })
  await context.route(/supabase\.co|cloudfront\.net/, async (route) => {
    const req = route.request(), url = new URL(req.url()), p = url.pathname
    const json = (b, st = 200) => route.fulfill({ status: st, contentType: "application/json", body: JSON.stringify(b) })
    const body = () => { try { return JSON.parse(req.postData() ?? "{}") } catch { return {} } }
    if (!url.hostname.includes("supabase")) return route.fulfill({ status: 404, body: "" })
    if (p === "/auth/v1/user") return json(USER)
    if (p.startsWith("/auth/")) return json(SESSION)
    if (p.includes("/rpc/")) {
      const fn = p.split("/rpc/")[1]; const b = body(); calls.push({ fn, body: b })
      if (fn === "finance_report") return json([{ previsto_centavos: 0, recebido_centavos: 0, custos_centavos: 150000, comissoes_provisionadas_centavos: 0, comissoes_liquidadas_centavos: 0, resultado_estimado_centavos: -150000, despesas_a_pagar_centavos: 47990 }])
      if (fn === "create_expense") { state.entries.push(conta(b.p_id, b.p_descricao, b.p_categoria, b.p_valor_centavos, b.p_vencimento, { forma: b.p_forma, situacao: b.p_pago_em ? "paga" : "pendente", pago_em: b.p_pago_em, valor_pago_centavos: b.p_pago_em ? b.p_valor_centavos : null })); return json(b.p_id) }
      if (fn === "create_expenses") { for (const i of b.p_items) state.entries.push(conta(i.id, i.descricao, i.categoria, i.valor_centavos, i.vencimento, { forma: i.forma })); return json(b.p_items.length) }
      if (fn === "update_expense") { const e = state.entries.find((x) => x.id === b.p_id); Object.assign(e, { descricao: b.p_descricao, categoria: b.p_categoria, valor_centavos: b.p_valor_centavos, data: b.p_vencimento, forma: b.p_forma }); return json(null) }
      if (fn === "pay_expense") { const e = state.entries.find((x) => x.id === b.p_id); Object.assign(e, { situacao: "paga", pago_em: b.p_pago_em, valor_pago_centavos: b.p_valor_pago_centavos, forma: b.p_forma }); return json(null) }
      if (fn === "revert_expense_payment") { const e = state.entries.find((x) => x.id === b.p_id); Object.assign(e, { situacao: "pendente", pago_em: null, valor_pago_centavos: null }); return json(null) }
      return json(fn === "sync_expense_recurrences" ? 0 : null)
    }
    const t = p.replace("/rest/v1/", "")
    if (t.startsWith("memberships")) return json([{ papel: "proprietario", barbershops: SHOP }])
    if (t.startsWith("expense_recurrences")) return json([{ id: "r1", descricao: "Aluguel", categoria: "Aluguel", valor_centavos: 150000, frequencia: "mensal", inicio: d(-6) }])
    if (t.startsWith("expense_events")) return json([{ id: "ev1", tipo: "pagamento", pago_em: d(-6), valor_centavos: 150000, forma: "pix", criado_em: new Date().toISOString() }])
    if (t.startsWith("expenses")) return json(state.entries)
    return json(req.headers()["accept"]?.includes("vnd.pgrst.object") ? null : [])
  })
  return { browser, page, errors, calls, state }
}

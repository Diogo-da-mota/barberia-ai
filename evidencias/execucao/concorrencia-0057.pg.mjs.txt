// ---- conexões como o PostgREST: transação, papel authenticated e claims locais ----
const agora = () => Math.floor(Date.now() / 1000)
async function conectar() { const c = new pg.Client(cfg); await c.connect(); c.pid = (await c.query("select pg_backend_pid() p")).rows[0].p; return c }
async function como(c, uid, sql, params = [], { amr = [{ method: "password", timestamp: agora() }], aberta = false } = {}) {
  await c.query("begin")
  await c.query("select set_config('request.jwt.claims', $1, true), set_config('request.jwt.claim.sub', $2, true)", [JSON.stringify({ sub: uid, role: "authenticated", amr }), uid])
  await c.query("set local role authenticated")
  try {
    const r = await c.query(sql, params)
    if (!aberta) await c.query("commit")
    return r.rows
  } catch (e) {
    await c.query("rollback").catch(() => {})
    throw e
  }
}
const tentar = (p) => p.then((r) => ({ ok: r }), (e) => ({ erro: e.message, code: e.code }))
async function esperandoTrava(c, ms = 5000) {
  for (const fim = Date.now() + ms; Date.now() < fim; await new Promise((r) => setTimeout(r, 25))) {
    const [r] = await q("select wait_event_type t from pg_stat_activity where pid = $1", [c.pid])
    if (r?.t === "Lock") return true
  }
  return false
}

async function open({ width, height, deleteError = "reauth_required" }) {
  const browser = await chromium.launch({ channel: "chrome", headless: true })
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 768 })
  const page = await context.newPage()
  const errors = [], calls = []
  let deletes = 0
  page.on("pageerror", (e) => errors.push(e.message))
  page.on("console", (m) => { if (m.type() === "error" && !/Failed to load resource|status of 400/.test(m.text())) errors.push(m.text()) })
  await context.addInitScript(({ s }) => { localStorage.setItem("barberia:theme", "dark"); localStorage.setItem("barberia-auth-painel", JSON.stringify(s)) }, { s: session(now() - 7200) })
  await context.route(/supabase\.co|cloudfront\.net/, async (route) => {
    const req = route.request(), url = new URL(req.url()), p = url.pathname
    const json = (b, st = 200) => route.fulfill({ status: st, contentType: "application/json", body: JSON.stringify(b) })
    const body = () => { try { return JSON.parse(req.postData() ?? "{}") } catch { return {} } }
    if (!url.hostname.includes("supabase")) return route.fulfill({ status: 404, body: "" })
    if (p === "/auth/v1/user") return json(USER)
    if (p === "/auth/v1/token") { calls.push({ fn: `token:${url.searchParams.get("grant_type")}`, body: body() }); return json(session(now())) }
    if (p.startsWith("/auth/")) { calls.push({ fn: p }); return json({}) }
    if (p.includes("/rpc/")) {
      const fn = p.split("/rpc/")[1], b = body(); calls.push({ fn, body: b })
      if (fn === "get_leave_summary") return json(b.p_barbershop_id === "loja-2"
        ? { papel: "barbeiro", pode_sair: true, reservas_futuras: 1, comissoes_a_receber_centavos: 2400, google_agenda: true }
        : { papel: "barbeiro", pode_sair: true, reservas_futuras: 0, comissoes_a_receber_centavos: 0, google_agenda: false })
      if (fn === "leave_barbershop") return json({ status: "saiu", reservas_futuras: 1 })
      if (fn === "delete_my_account") {
        deletes++
        if (deleteError === "owner_cannot_delete" || deletes === 1) return json({ code: "P0001", details: null, hint: null, message: deleteError }, 400)
        return route.fulfill({ status: 204, body: "" })
      }
      if (fn === "request_account_deletion") return json({ id: "p1", status: "pendente", bloqueios: ["proprietario"], criado_em: new Date().toISOString() })
      return json(null)
    }
    const t = p.replace("/rest/v1/", "")
    if (t.startsWith("memberships")) return json(SHOPS)
    return json(req.headers()["accept"]?.includes("vnd.pgrst.object") ? null : [])
  })
  return { browser, page, errors, calls }
}

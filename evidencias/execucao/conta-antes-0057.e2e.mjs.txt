const b64 = (o) => Buffer.from(JSON.stringify(o)).toString("base64url")
const now = () => Math.floor(Date.now() / 1000)
const USER = { id: "00000000-0000-4000-8000-0000000000b1", aud: "authenticated", role: "authenticated", email: "carlos@example.test", app_metadata: { provider: "email", providers: ["email"] }, user_metadata: { full_name: "Carlos Fictício" }, identities: [], created_at: new Date().toISOString() }
const JWT = `${b64({ alg: "none" })}.${b64({ sub: USER.id, exp: now() + 3600, role: "authenticated", amr: [{ method: "password", timestamp: now() - 7200 }] })}.teste`

]
const NOVAS = new Set(["get_leave_summary", "leave_barbershop", "request_account_deletion"])

async function open({ width, height, dono = false }) {
  const browser = await chromium.launch({ channel: "chrome", headless: true })
  const context = await browser.newContext({ viewport: { width, height } })
  const page = await context.newPage()
  const errors = [], calls = []
  page.on("pageerror", (e) => errors.push(e.message))
  page.on("console", (m) => { if (m.type() === "error" && !/Failed to load resource|status of 4\d\d/.test(m.text())) errors.push(m.text()) })
  await context.addInitScript(({ s }) => { localStorage.setItem("barberia:theme", "dark"); localStorage.setItem("barberia-auth-painel", JSON.stringify(s)) }, { s: SESSION })
  await context.route(/supabase\.co|cloudfront\.net/, async (route) => {
    const req = route.request(), url = new URL(req.url()), p = url.pathname
    const json = (b, st = 200) => route.fulfill({ status: st, contentType: "application/json", body: JSON.stringify(b) })
    if (!url.hostname.includes("supabase")) return route.fulfill({ status: 404, body: "" })
    if (p === "/auth/v1/user") return json(USER)
    if (p.startsWith("/auth/")) { calls.push(p); return json({}) }
    if (p.includes("/rpc/")) {
      const fn = p.split("/rpc/")[1]; calls.push(fn)
      if (NOVAS.has(fn)) return json({ code: "PGRST202", details: null, hint: null, message: `Could not find the function public.${fn}` }, 404)
      if (fn === "delete_my_account") return dono ? json({ code: "P0001", details: null, hint: null, message: "owner_cannot_delete" }, 400) : route.fulfill({ status: 204, body: "" })
      return json(null)
    }
    const t = p.replace("/rest/v1/", "")
    if (t.startsWith("account_deletion_requests")) { calls.push("tabela:account_deletion_requests"); return json({ code: "PGRST205", details: null, hint: null, message: "Could not find the table" }, 404) }
    if (t.startsWith("memberships")) return json(dono ? SHOPS : [SHOPS[0]])
    return json(req.headers()["accept"]?.includes("vnd.pgrst.object") ? null : [])
  })
  return { browser, page, errors, calls }
}

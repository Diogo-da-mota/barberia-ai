/** real=true: fala com o Supabase real. real=false: toda chamada ao Supabase é interceptada. */
async function open({ width = 1440, height = 900, real = false, session = false, theme = "dark", fakeSession = null, mocks = {}, rest = "unavailable" }) {
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 768 })
  const page = await context.newPage()
  const log = { pageErrors: [], consoleErrors: [], requests: [] }
  page.on("pageerror", (e) => log.pageErrors.push(e.message))
  page.on("console", (m) => { if (m.type() === "error") log.consoleErrors.push(m.text()) })
  page.on("request", (r) => { if (r.url().includes("supabase.co")) log.requests.push({ url: r.url(), method: r.method(), body: r.postData() }) })

  const stored = fakeSession ?? (session ? SESSION : null)
  await context.addInitScript(({ theme, stored, key }) => {
    localStorage.setItem("barberia:theme", theme)
    if (stored && !sessionStorage.getItem("__seeded")) {
      sessionStorage.setItem("__seeded", "1")
      localStorage.setItem(key, JSON.stringify(stored))
    }
  }, { theme, stored, key: TOKEN_KEY })

  if (!real) {
    await context.route(/supabase\.co/, async (route) => {
      const url = route.request().url()
      const method = route.request().method()
      if (url.includes("/rest/v1/")) {
        return rest === "unavailable"
          ? route.fulfill({ status: 404, contentType: "application/json", body: JSON.stringify({ code: "PGRST205", message: "tabela ausente" }) })
          : route.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify(rest) })
      }
      for (const [pattern, reply] of Object.entries(mocks)) {
        if (url.includes(pattern)) {
          const r = typeof reply === "function" ? reply(route.request()) : reply
          return route.fulfill({ status: r.status ?? 200, contentType: "application/json", body: JSON.stringify(r.body ?? {}) })
        }
      }
      if (url.includes("/auth/v1/logout")) return route.fulfill({ status: 204, body: "" })
      if (url.includes("/auth/v1/user")) {
        return method === "GET" || method === "PUT"
          ? route.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify(USER) })
          : route.abort()
      }
      return route.abort()
    })
  }
  return { page, context, log }
}

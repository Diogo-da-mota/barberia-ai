import { chromium } from "playwright-core"
import { mkdirSync } from "node:fs"

const BASE = process.env.BASE ?? "http://localhost:4173"
const OUT = process.argv[2]
mkdirSync(OUT, { recursive: true })
let ok = 0, bad = 0
const check = (name, pass, detail = "") => { pass ? ok++ : bad++; console.log(`${pass ? "PASS" : "FAIL"}  ${name}${detail ? `  (${detail})` : ""}`) }

const b64 = (o) => Buffer.from(JSON.stringify(o)).toString("base64url")
const exp = () => Math.floor(Date.now() / 1000) + 3600
const USER = { id: "00000000-0000-4000-8000-000000000001", aud: "authenticated", role: "authenticated", email: "teste@example.test", app_metadata: {}, user_metadata: { full_name: "Usuária Teste" }, identities: [{ id: "x" }], created_at: "2026-01-01T00:00:00Z" }
const JWT = `${b64({ alg: "none" })}.${b64({ sub: USER.id, exp: exp(), role: "authenticated" })}.teste`
const SESSION = { access_token: JWT, refresh_token: "teste", token_type: "bearer", expires_in: 3600, expires_at: exp(), user: USER }
const shop = (papel) => ({ papel, barbershops: { id: "loja-1", nome: "Barbearia Fictícia", slug: "barbearia-ficticia", endereco: "Rua de Teste", status: "ativo", fuso: "America/Sao_Paulo" } })
const now = Date.now()
const BOOKING = { id: "b1", barbershop_id: "loja-1", professional_id: "p1", customer_id: "outro", service_id: "s1", servico_nome: "Corte", preco_centavos: 6000, duracao_min: 40, inicio: new Date(now - 60_000).toISOString(), fim: new Date(now + 39 * 60_000).toISOString(), status: "confirmado", cliente_nome: "Cliente Um", cliente_whatsapp: "11912345678", professionals: { nome: "Ana" }, barbershops: { nome: "Barbearia Fictícia", fuso: "America/Sao_Paulo", slug: "barbearia-ficticia" } }
const TEAM = [{ id: "p1", nome: "Ana", ativo: true, professional_services: [{ service_id: "s1" }] }]

const browser = await chromium.launch({ channel: "chrome", headless: true })
async function open(papel, width = 1440, height = 900) {
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 768 })
  const page = await context.newPage()
  const errors = []
  page.on("pageerror", (e) => errors.push(e.message))
  page.on("console", (m) => { if (m.type() === "error" && !/Failed to load resource/.test(m.text())) errors.push(m.text()) })
  await context.addInitScript(({ s }) => { localStorage.setItem("barberia:theme", "dark"); if (!sessionStorage.getItem("__s")) { sessionStorage.setItem("__s", "1"); localStorage.setItem("sb-hbkbdqctelembzqiptpi-auth-token", JSON.stringify(s)) } }, { s: SESSION })
  await context.route(/supabase\.co/, async (route) => {
    const p = new URL(route.request().url()).pathname
    const json = (b, st = 200) => route.fulfill({ status: st, contentType: "application/json", body: JSON.stringify(b) })
    if (p === "/auth/v1/user") return json(USER)
    if (p.endsWith("/rpc/get_team_links")) return json([])
    if (p.includes("/rpc/")) return json(null)
    if (p.startsWith("/rest/v1/memberships")) return json(papel === "barbeiro" && route.request().url().includes("papel=eq.barbeiro") ? [] : [shop(papel)])
    if (p.startsWith("/rest/v1/bookings")) return json(papel === "cliente" ? [] : [BOOKING])
    if (p.startsWith("/rest/v1/occupancies")) return json([])
    if (p.startsWith("/rest/v1/professionals")) return json(TEAM)
    if (p.startsWith("/rest/v1/services")) return json([{ id: "s1", nome: "Corte", descricao: null, preco_centavos: 6000, duracao_min: 40, ativo: true }])
    if (p.startsWith("/rest/v1/working_hours")) return json([])
    if (p.startsWith("/rest/v1/invites")) return json([])
    return route.abort()
  })
  return { page, context, errors }
}

const ow = (page) => page.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth)
for (const [w, h] of [[390, 844], [768, 1024], [1440, 900]]) {
  const { page, context, errors } = await open("proprietario", w, h)
  await page.goto(BASE + "/dashboard/mais", { waitUntil: "load" })
  await page.getByRole("link", { name: "Clientes" }).click()
  await page.waitForURL(BASE + "/dashboard/clientes")
  await page.getByText("Cliente Um").first().waitFor()
  check(`${w}px: Mais leva a /dashboard/clientes e lista o cliente`, await page.getByRole("link", { name: /WhatsApp 11912345678/ }).isVisible())
  check(`${w}px: sem scroll horizontal`, (await ow(page)) <= 0)
  await page.getByLabel("Buscar por nome ou WhatsApp").fill("zzz")
  check(`${w}px: busca sem resultado mostra a mensagem`, await page.getByText("Nenhum cliente encontrado para essa busca.").isVisible())
  await page.getByLabel("Buscar por nome ou WhatsApp").fill("cliente um")
  check(`${w}px: busca por nome encontra o cliente`, await page.getByText("Cliente Um").first().isVisible())
  if (w < 768) { const small = await page.evaluate(() => [...document.querySelectorAll("main a, main button, main input")].filter((e) => e.offsetParent && e.offsetHeight < 44).map((e) => e.tagName + ":" + e.offsetHeight)); check(`${w}px: alvos de toque >= 44px`, small.length === 0, small.join(",")) }
  check(`${w}px: sem exceção de aplicação`, errors.length === 0, errors.join(" | ").slice(0, 100))
  if (w !== 768) await page.screenshot({ path: `${OUT}/clientes-${w}.png` })
  await context.close()
}
{
  const { page, context } = await open("barbeiro")
  await page.goto(BASE + "/dashboard/clientes", { waitUntil: "load" })
  await page.getByText("Cliente Um").first().waitFor()
  check("barbeiro: acessa /dashboard/clientes", true)
  await context.close()
}
{
  const { page, context } = await open("cliente")
  await page.goto(BASE + "/dashboard/clientes", { waitUntil: "load" })
  await page.getByText("Crie sua barbearia primeiro").waitFor()
  check("cliente: não acessa /dashboard/clientes", true)
  await context.close()
}
await browser.close()
console.log(`RESUMO: ${ok}/${ok + bad} aprovadas`)
process.exit(bad ? 1 : 0)

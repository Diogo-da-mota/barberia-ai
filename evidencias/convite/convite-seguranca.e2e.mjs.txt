async function open(person, { width = 1440, height = 900 } = {}) {
  const browser = await chromium.launch({ channel: "chrome", headless: true })
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 768 })
  const page = await context.newPage()
  const errors = []
  page.on("pageerror", (e) => errors.push(e.message))
  await context.addInitScript(({ s }) => { localStorage.setItem("barberia:theme", "dark"); if (s) localStorage.setItem("barberia-auth-painel", JSON.stringify(s)) }, { s: person ? sessionOf(person) : null })
  await context.route(/supabase\.co|cloudfront\.net/, async (route) => {
    const req = route.request(), url = new URL(req.url()), p = url.pathname
    const json = (b, st = 200) => route.fulfill({ status: st, contentType: "application/json", body: JSON.stringify(b) })
    if (!url.hostname.includes("supabase")) return route.fulfill({ status: 404, body: "" })
    supabaseOrigin = url.origin
    const sub = subOf(req), who = Object.values(P).find((x) => x.id === sub)
    if (p === "/auth/v1/user") return who ? json(userOf(who)) : json({ message: "no session" }, 401)
    if (p === "/auth/v1/logout") return route.fulfill({ status: 204, body: "" })
    if (p === "/auth/v1/authorize") { authorizeUrls.push(req.url()); return route.fulfill({ status: 200, contentType: "text/html", body: "<p>Google (simulado)</p>" }) }
    if (p.startsWith("/auth/")) return json(who ? sessionOf(who) : {})
    if (p.includes("/rpc/")) {
      const fn = p.split("/rpc/")[1], body = JSON.parse(req.postData() ?? "{}")
      try {
        if (fn === "accept_invite") return json(await one(sub, "select accept_invite($1)", [body.p_token]))
        if (fn === "get_plan_usage") return json(await as(sub, "select * from get_plan_usage($1)", [body.p_barbershop_id]))
      } catch (e) { return json({ code: e.code ?? "P0001", message: e.message, details: null, hint: null }, 400) }
      return json([])
    }
    const t = p.replace("/rest/v1/", "")
    if (t === "memberships" && url.searchParams.get("profile_id")) return json(await as(sub, "select m.papel, json_build_object('id', b.id, 'nome', b.nome, 'slug', b.slug, 'endereco', null, 'status', 'ativo', 'fuso', 'America/Sao_Paulo') barbershops from memberships m join barbershops b on b.id = m.barbershop_id where m.profile_id = $1", [sub]))
    if (t === "professionals") return json(await as(sub, "select id, nome, ativo, avatar_path, '[]'::json professional_services from professionals where barbershop_id = $1 order by nome", [loja]))
    return json(req.headers()["accept"]?.includes("vnd.pgrst.object") ? null : [])
  })
  return { browser, page, errors }
}

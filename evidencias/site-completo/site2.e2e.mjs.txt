const browser = await chromium.launch({ channel: "chrome", headless: true })

async function open({ width = 1440, height = 900, logged = true }) {
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 768 })
  const page = await context.newPage()
  const log = { errors: [], calls: [] }
  page.on("pageerror", (e) => log.errors.push(e.message))
  page.on("console", (m) => { if (m.type() === "error" && !/Failed to load resource/.test(m.text())) log.errors.push(m.text()) })
  await context.addInitScript(({ session, logged }) => {
    localStorage.setItem("barberia:theme", "dark")
    if (logged && !sessionStorage.getItem("__seeded")) { sessionStorage.setItem("__seeded", "1"); localStorage.setItem("sb-hbkbdqctelembzqiptpi-auth-token", JSON.stringify(session)) }
  }, { session: SESSION, logged })
  await context.route(/supabase\.co/, async (route) => {
    const req = route.request(); const url = new URL(req.url()); const p = url.pathname
    const json = (body, status = 200) => route.fulfill({ status, contentType: "application/json", body: JSON.stringify(body) })
    const body = req.postData() ? JSON.parse(req.postData()) : null
    if (req.method() !== "GET") log.calls.push({ path: p, body })
    if (p === "/auth/v1/user") return json(USER)
    if (p === "/auth/v1/logout") return route.fulfill({ status: 204, body: "" })
    if (p.endsWith("/rpc/get_public_shop")) return json(PUBLIC)
    if (p.endsWith("/rpc/get_availability")) { const d = body.p_data; return json([{ prof_id: "p1", prof_nome: "Ana", slot_inicio: `${d}T12:00:00+00:00` }, { prof_id: "p1", prof_nome: "Ana", slot_inicio: `${d}T12:15:00+00:00` }, { prof_id: "p2", prof_nome: "Bia", slot_inicio: `${d}T13:00:00+00:00` }]) }
    if (p.endsWith("/rpc/create_booking")) return json("novo")
    if (p.endsWith("/rpc/create_invite")) return json("tok-abc")
    if (p.includes("/rpc/")) return json(null)
    if (p.startsWith("/rest/v1/memberships")) return json([SHOP])
    if (p.startsWith("/rest/v1/bookings")) return json(BOOKINGS)
    if (p.startsWith("/rest/v1/occupancies")) return json([])
    if (p.startsWith("/rest/v1/professionals")) return json(TEAM)
    if (p.startsWith("/rest/v1/services")) return json(PUBLIC.services)
    if (p.startsWith("/rest/v1/working_hours")) return json(HOURS)
    if (p.startsWith("/rest/v1/invites")) return json([])
    return route.abort()
  })
  return { page, context, log }
}

import { chromium } from "playwright-core"
import { mkdirSync } from "node:fs"

const BASE = process.env.BASE ?? "http://localhost:4173"
const OUT = process.argv[2]
mkdirSync(OUT, { recursive: true })
let ok = 0, bad = 0
const check = (name, pass, detail = "") => { pass ? ok++ : bad++; console.log(`${pass ? "PASS" : "FAIL"}  ${name}${detail ? `  (${detail})` : ""}`) }

const b64 = (o) => Buffer.from(JSON.stringify(o)).toString("base64url")
const exp = () => Math.floor(Date.now() / 1000) + 3600
const USER = { id: "00000000-0000-4000-8000-000000000001", aud: "authenticated", role: "authenticated", email: "teste@example.test", app_metadata: {}, user_metadata: { full_name: "Usuária Teste" }, identities: [{ id: "x" }], created_at: "2026-01-01T00:00:00Z" }
const JWT = `${b64({ alg: "none" })}.${b64({ sub: USER.id, exp: exp(), role: "authenticated" })}.teste`
const SESSION = { access_token: JWT, refresh_token: "teste", token_type: "bearer", expires_in: 3600, expires_at: exp(), user: USER }
const shop = (papel) => ({ papel, barbershops: { id: "loja-1", nome: "Barbearia Fictícia", slug: "barbearia-ficticia", endereco: "Rua de Teste", status: "ativo", fuso: "America/Sao_Paulo" } })
const now = Date.now()
const BOOKING = { id: "b1", barbershop_id: "loja-1", professional_id: "p1", customer_id: "outro", service_id: "s1", servico_nome: "Corte", preco_centavos: 6000, duracao_min: 40, inicio: new Date(now - 60_000).toISOString(), fim: new Date(now + 39 * 60_000).toISOString(), status: "confirmado", cliente_nome: "Cliente Um", cliente_whatsapp: "11912345678", professionals: { nome: "Ana" }, barbershops: { nome: "Barbearia Fictícia", fuso: "America/Sao_Paulo", slug: "barbearia-ficticia" } }
const TEAM = [{ id: "p1", nome: "Ana", ativo: true, professional_services: [{ service_id: "s1" }] }]

const browser = await chromium.launch({ channel: "chrome", headless: true })
async function open(papel, width = 1440, height = 900) {
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 768 })
  const page = await context.newPage()
  const errors = []
  page.on("pageerror", (e) => errors.push(e.message))
  page.on("console", (m) => { if (m.type() === "error" && !/Failed to load resource/.test(m.text())) errors.push(m.text()) })
  await context.addInitScript(({ s }) => { localStorage.setItem("barberia:theme", "dark"); if (!sessionStorage.getItem("__s")) { sessionStorage.setItem("__s", "1"); localStorage.setItem("sb-hbkbdqctelembzqiptpi-auth-token", JSON.stringify(s)) } }, { s: SESSION })
  await context.route(/supabase\.co/, async (route) => {
    const p = new URL(route.request().url()).pathname
    const json = (b, st = 200) => route.fulfill({ status: st, contentType: "application/json", body: JSON.stringify(b) })
    if (p === "/auth/v1/user") return json(USER)
    if (p.endsWith("/rpc/get_team_links")) return json([])
    if (p.includes("/rpc/")) return json(null)
    if (p.startsWith("/rest/v1/memberships")) return json(papel === "barbeiro" && route.request().url().includes("papel=eq.barbeiro") ? [] : [shop(papel)])
    if (p.startsWith("/rest/v1/bookings")) return json(papel === "cliente" ? [] : [BOOKING])
    if (p.startsWith("/rest/v1/occupancies")) return json([])
    if (p.startsWith("/rest/v1/professionals")) return json(TEAM)
    if (p.startsWith("/rest/v1/services")) return json([{ id: "s1", nome: "Corte", descricao: null, preco_centavos: 6000, duracao_min: 40, ativo: true }])
    if (p.startsWith("/rest/v1/working_hours")) return json([])
    if (p.startsWith("/rest/v1/invites")) return json([])
    return route.abort()
  })
  return { page, context, errors }
}
const count = (page, role, name) => page.getByRole(role, { name }).count()

for (const papel of (process.env.PAPEIS ?? "proprietario,gerente,barbeiro,cliente").split(",")) {
  const { page, context, errors } = await open(papel)
  // Agenda
  await page.goto(BASE + "/dashboard/agendamentos", { waitUntil: "load" })
  if (papel === "cliente") {
    await page.getByText("Crie sua barbearia primeiro").waitFor()
    check(`${papel}: não acessa a agenda da equipe`, true)
  } else {
    await page.getByText("Cliente Um").waitFor()
    const manage = papel !== "barbeiro"
    check(`${papel}: ${manage ? "vê Cancelar e Bloquear horário" : "só conclui e marca falta"}`, (await count(page, "button", /Cancelar reserva de/)) === (manage ? 1 : 0) && (await count(page, "button", "Bloquear horário")) === (manage ? 1 : 0) && (await count(page, "button", "Concluir")) === 1 && (await count(page, "button", "Faltou")) === 1)
  }
  // Equipe
  await page.goto(BASE + "/dashboard/equipe", { waitUntil: "load" })
  if (papel === "cliente") await page.getByText("Crie sua barbearia primeiro").waitFor()
  else await page.getByText("Ana").first().waitFor()
  check(`${papel}: ${papel === "proprietario" || papel === "gerente" ? "edita a equipe" : "não edita a equipe"}`, (await count(page, "button", "Novo profissional")) + (await count(page, "button", "Editar Ana")) === (papel === "proprietario" || papel === "gerente" ? 1 : 0) || (papel === "proprietario" || papel === "gerente") && (await count(page, "button", "Editar Ana")) === 1)
  // Horários
  await page.goto(BASE + "/dashboard/horarios", { waitUntil: "load" })
  await page.getByRole("heading", { level: 1 }).waitFor()
  await page.waitForTimeout(300)
  const editsHours = (await page.getByRole("button", { name: "Salvar horários" }).count()) === 1
  check(`${papel}: ${papel === "proprietario" || papel === "gerente" ? "edita horários" : "não edita horários"}`, editsHours === (papel === "proprietario" || papel === "gerente"))
  // Mais (convites)
  await page.goto(BASE + "/dashboard/mais", { waitUntil: "load" })
  await page.getByRole("link", { name: /Minhas reservas/ }).waitFor()
  if (papel === "proprietario" || papel === "gerente") await page.getByRole("button", { name: "Gerar link do convite" }).waitFor()
  else await page.getByText("Seu papel").waitFor()
  const invites = (await count(page, "button", "Gerar link do convite")) === 1
  const ownerRadio = (await count(page, "radio", "Gerente")) === 1
  check(`${papel}: convites ${papel === "proprietario" ? "com opção de gerente" : papel === "gerente" ? "só para barbeiro" : "indisponíveis"}`, papel === "proprietario" ? invites && ownerRadio : papel === "gerente" ? invites && !ownerRadio : !invites)
  // Painel
  await page.goto(BASE + "/dashboard", { waitUntil: "load" })
  await page.getByRole("heading", { level: 1 }).waitFor()
  await page.waitForTimeout(300)
  const heading = papel === "cliente" ? "Seus próximos agendamentos" : "Próximos atendimentos"
  check(`${papel}: painel mostra "${heading}"`, await page.getByRole("heading", { name: heading }).isVisible())
  check(`${papel}: indicadores ${papel === "cliente" ? "ocultos" : "visíveis"}`, (await page.getByRole("region", { name: "Indicadores" }).count()) === (papel === "cliente" ? 0 : 1))
  check(`${papel}: sem exceção de aplicação`, errors.length === 0, errors.join(" | ").slice(0, 120))
  await context.close()
}

// Visões da agenda e responsividade
for (const [w, h] of (process.env.PAPEIS ? [] : [[390, 844], [768, 1024], [1440, 900]])) {
  const { page, context } = await open("proprietario", w, h)
  await page.goto(BASE + "/dashboard/agendamentos", { waitUntil: "load" })
  await page.getByText("Cliente Um").waitFor()
  await page.getByRole("button", { name: "Semana" }).click()
  await page.getByRole("region", { name: /\d/ }).first().waitFor()
  const ow = await page.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth)
  check(`${w}px agenda em semana: sem scroll horizontal`, ow <= 0)
  if (w === 390) await page.screenshot({ path: `${OUT}/agenda-semana-390.png` })
  await page.getByRole("button", { name: "Mês" }).click()
  await page.getByRole("button", { name: /, 1 item$/ }).waitFor()
  const om = await page.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth)
  check(`${w}px agenda em mês: sem scroll horizontal`, om <= 0)
  const small = await page.evaluate(() => [...document.querySelectorAll("main button")].filter((e) => e.offsetParent && e.offsetHeight < 44 && innerWidth < 768).map((e) => e.textContent.slice(0, 10) + ":" + e.offsetHeight))
  if (w < 768) check(`${w}px agenda: alvos de toque >= 44px`, small.length === 0, small.slice(0, 3).join(", "))
  await page.screenshot({ path: `${OUT}/agenda-mes-${w}.png` })
  await page.getByRole("button", { name: /, 1 item$/ }).click()
  await page.getByText("Cliente Um").waitFor()
  check(`${w}px mês: clicar no dia abre a visão do dia`, (await page.getByRole("button", { name: "Dia", exact: true }).getAttribute("aria-pressed")) === "true")
  await context.close()
}

await browser.close()
console.log(`RESUMO: ${ok}/${ok + bad} aprovadas`)
process.exit(bad ? 1 : 0)

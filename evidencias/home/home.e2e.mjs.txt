const RESPONSES = {
  unavailable: { status: 404, body: { code: "PGRST205", message: "tabela ausente" } },
  empty: { status: 200, body: [] },
  populated: { status: 200, body: FICTICIO },

const browser = await chromium.launch({ channel: "chrome", headless: true })

async function open({ width, height = 900, state, session = true, theme = "dark", path = "/" }) {
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 768 })
  const page = await context.newPage()
  const log = { pageErrors: [], consoleErrors: [], outgoing: [] }
  page.on("pageerror", (e) => log.pageErrors.push(e.message))
  page.on("console", (m) => { if (m.type() === "error") log.consoleErrors.push(m.text()) })

  await page.addInitScript(({ session, theme, sessionData }) => {
    localStorage.setItem("barberia:theme", theme)
    if (session && !sessionStorage.getItem("__logged_out")) {
      localStorage.setItem("sb-hbkbdqctelembzqiptpi-auth-token", JSON.stringify(sessionData))
    }
  }, { session, theme, sessionData: SESSION })

  if (session) {
    let mode = state
    await page.route(/\/rest\/v1\//, async (route) => {
      log.outgoing.push(route.request().url())
      if (mode === "loading") return // nunca responde
      const { status, body } = RESPONSES[mode]
      await route.fulfill({ status, contentType: "application/json", body: JSON.stringify(body) })
    })
    await page.route(/\/auth\/v1\//, async (route) => {
      await page.evaluate(() => sessionStorage.setItem("__logged_out", "1")).catch(() => {})
      await route.fulfill({ status: 204, body: "" })
    })
    page.setMode = (m) => { mode = m }
  }

  await page.goto(BASE + path, { waitUntil: "load" })
  return { page, context, log }
}

const noOverflow = (page) => page.evaluate(() => {
  const main = document.querySelector("main")
  return {
    doc: document.documentElement.scrollWidth - document.documentElement.clientWidth,
    main: main ? main.scrollWidth - main.clientWidth : 0,
  }
})

// ---------- 4. Responsividade nos breakpoints do DS ----------
for (const width of [320, 360, 390, 768, 1024, 1440]) {
  for (const state of ["populated", "unavailable"]) {
    const { page, context } = await open({ width, state })
    await page.getByRole("heading", { level: 1 }).waitFor()
    await page.getByRole("heading", { name: "Suas barbearias" }).waitFor()
    await page.getByRole("region", { name: "Indicadores" }).waitFor()
    const over = await noOverflow(page)
    check(`${width}px/${state}: sem scroll horizontal`, over.doc <= 0 && over.main <= 0, JSON.stringify(over))
    const layout = await page.evaluate(() => {
      const vis = (el) => !!el && getComputedStyle(el).display !== "none" && el.offsetWidth > 0
      const aside = document.querySelector("aside")
      const bottom = document.querySelector("nav.fixed")
      const clipped = [...document.querySelectorAll("nav.fixed a span:last-child")].filter((s) => s.scrollWidth > s.parentElement.clientWidth).length
      const main = document.querySelector("main")
      main.scrollTop = main.scrollHeight
      const last = main.firstElementChild.getBoundingClientRect()
      return { aside: vis(aside), asideW: aside.offsetWidth, bottom: vis(bottom), clipped, lastBottom: last.bottom, navTop: vis(bottom) ? bottom.getBoundingClientRect().top : null }
    })
    if (width < 768) {
      check(`${width}px/${state}: barra inferior, sem sidebar`, layout.bottom && !layout.aside)
      check(`${width}px/${state}: rótulos da barra sem corte`, layout.clipped === 0, `${layout.clipped}`)
      check(`${width}px/${state}: nav não cobre o fim do conteúdo`, layout.lastBottom <= layout.navTop, `${Math.round(layout.lastBottom)}<=${Math.round(layout.navTop)}`)
      const small = await page.evaluate(() => [...document.querySelectorAll("nav.fixed a, header a, header button, main a, main button")].filter((e) => e.offsetHeight > 0 && e.offsetHeight < 44).map((e) => (e.textContent || e.getAttribute("aria-label")) + ":" + e.offsetHeight))
      check(`${width}px/${state}: alvos de toque >= 44px`, small.length === 0, small.join(", "))
    } else if (width < 1024) {
      check(`${width}px/${state}: sidebar compacta 64px`, layout.aside && layout.asideW === 64 && !layout.bottom, `${layout.asideW}`)
    } else {
      check(`${width}px/${state}: sidebar completa`, layout.aside && layout.asideW === 240 && !layout.bottom, `${layout.asideW}`)
    }
    if ([390, 768, 1024, 1440].includes(width)) {
      await page.evaluate(() => { document.querySelector("main").scrollTop = 0 })
      await page.screenshot({ path: `${OUT}/home-${state === "unavailable" ? "estado-atual" : "dados-ficticios"}-${width}-dark.png`, fullPage: false })
    }
    await context.close()
  }
}
